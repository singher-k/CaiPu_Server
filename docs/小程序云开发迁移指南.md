# 小程序云开发迁移指南

## 迁移原因

原方案使用独立服务器 + HTTP API，存在以下问题：
- 微信体验版/正式版域名校验限制
- 需要配置 HTTPS 和备案
- 服务器维护成本

云开发方案优势：
- ✅ **免备案** - 无需域名备案
- ✅ **免HTTPS** - 自动支持 HTTPS
- ✅ **免运维** - 微信托管服务器
- ✅ **免费额度** - 个人小程序足够使用
- ✅ **天然打通** - 与小程序无缝集成

## 一、修改 app.js

```javascript
// app.js

App({
  onLaunch() {
    // 初始化云开发环境
    if (!wx.cloud) {
      console.error('请使用 2.2.3 或以上的基础库以使用云能力')
    } else {
      wx.cloud.init({
        // env 参数说明：
        // 1. 请替换为你的云开发环境ID
        // 2. 格式：环境ID（如：caipu-xxxxx）
        // 3. 可以在微信开发者工具云开发控制台查看
        env: 'your-env-id-here',
        traceUser: true,
      })
    }

    // 获取用户信息
    this.getUserInfo()
  },

  globalData: {
    userInfo: null,
    userId: null,  // 云开发返回的用户ID（_id）
    openid: null,  // 用户的openid
  },

  getUserInfo() {
    wx.getSetting({
      success: res => {
        if (res.authSetting['scope.userInfo']) {
          wx.getUserProfile({
            desc: '用于完善用户资料',
            success: res => {
              this.globalData.userInfo = res.userInfo
            }
          })
        }
      }
    })
  }
})
```

### 3.1 创建菜谱

```javascript
const api = require('../../utils/api')

async function createRecipe() {
  const recipeData = {
    title: '红烧肉',
    image: 'https://example.com/hongsaorou.jpg',
    category: '肉类',
    difficulty: '中等',
    time: '60分钟',
    ingredients: [
      { name: '五花肉', amount: '500g' },
      { name: '生抽', amount: '2勺' },
      { name: '老抽', amount: '1勺' }
    ],
    steps: [
      { step: 1, content: '五花肉切块焯水', image: '' },
      { step: 2, content: '炒糖色', image: '' },
      { step: 3, content: '加入五花肉翻炒', image: '' }
    ],
    userId: getApp().globalData.userId
  }

  const result = await api.createRecipe(recipeData)
  
  if (result.success) {
    wx.showToast({ title: '创建成功', icon: 'success' })

    wx.showToast({ title: result.message, icon: 'none' })
  }
}
```

## 四、修改登录调用代码

### 登录页面调用示例

```javascript
// pages/login/login.js
const api = require('../../utils/api')

Page({
  data: {
    canIUseGetUserProfile: false
  },

  onLoad() {
    if (wx.getUserProfile) {
      this.setData({ canIUseGetUserProfile: true })
    }
  },

  async login() {
    try {
      wx.showLoading({ title: '登录中...' })

      const loginRes = await new Promise((resolve, reject) => {
        wx.login({ success: resolve, fail: reject })
      })

      const userProfileRes = await new Promise((resolve, reject) => {
        wx.getUserProfile({
          desc: '用于完善用户资料',
          success: resolve,
          fail: reject
        })
      })

      const userInfo = {
        nickName: userProfileRes.userInfo.nickName,
        avatarUrl: userProfileRes.userInfo.avatarUrl
      }

      const result = await api.login(loginRes.code, userInfo)

      if (result.success) {
        wx.showToast({ title: '登录成功', icon: 'success' })
        getApp().globalData.userId = result.user.id
        setTimeout(() => {
          wx.switchTab({ url: '/pages/index/index' })
        }, 1500)
      } else {
        wx.showToast({ title: result.message, icon: 'none' })
      }
    } catch (error) {
      console.error('登录失败:', error)
      wx.showToast({ title: '登录失败', icon: 'none' })
    } finally {
      wx.hideLoading()
    }
  }
})
```
